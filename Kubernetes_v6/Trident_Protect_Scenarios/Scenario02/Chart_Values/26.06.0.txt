# Controls the ClusterRoles bound to the resourcebackup, resourcerestore, and
# resourcedelete ServiceAccounts. Most users should leave the defaults; this
# exists so security-sensitive deployments can scope these Jobs down.
#
# The mode is derived from which field you set; do NOT set rbac.mode (it has
# been removed and will fail loudly if present):
#
#   neither field set
#     -> "full" mode (default). Wildcard permissions on every API group and
#        resource. Backup gets get/list/use; restore gets every verb; delete
#        gets delete/use. Recommended unless you have a reason not to.
#
#   rbac.managedResources non-empty
#     -> "custom" mode. Render all three ClusterRoles from `managedResources`.
#        Backup, restore, and delete are driven by the same list so the three
#        roles cannot drift; the chart picks verbs per role:
#          resourcebackup  -> get, list (plus discovery non-resource URLs)
#          resourcerestore -> get, list, create, update, patch, delete (plus discovery)
#          resourcedelete  -> delete (plus discovery non-resource URLs)
#        Backup tolerates 401/403 by skipping the GVR (partial backup, logged),
#        but restore does NOT: if a backup contains a GVR missing from
#        `managedResources`, the restore Job fails. List every resource type
#        your apps use.
#
#        The chart auto-injects a read-only ([get,list]) baseline for the
#        cluster-scoped storage plumbing that backup/restore always need, so
#        you do NOT need to add these yourself:
#          - storage.k8s.io/storageclasses
#          - snapshot.storage.k8s.io/volumesnapshotclasses
#          - trident.netapp.io/tridentvolumes, tridentbackends
#        The resourcedelete role never gets the baseline; it only ever deletes
#        the resources you list.
#
#        The in-cluster HelmUpgrade Job (which needs */* due to RBAC escalation
#        prevention) is NOT rendered in custom mode. Run `helm upgrade`
#        externally with admin credentials instead.
#
#   rbac.existingClusterRoleName non-empty
#     -> "external" mode. Skip rendering all three ClusterRoles. Bind the
#        named ClusterRole to ALL three ServiceAccounts (resourcebackup,
#        resourcerestore, resourcedelete). Because the same role is reused for
#        all three Jobs, it must include every verb each Job needs - at minimum
#        the union of:
#          - backup:  get, list (plus discovery non-resource URLs)
#          - restore: get, list, create, update, patch, delete (plus discovery)
#          - delete:  delete (plus discovery non-resource URLs)
#        over every resource type your apps use. The chart still creates the
#        ServiceAccounts and ClusterRoleBindings.
#
# Setting BOTH fields is rejected (ambiguous).
rbac:
  # Single list of resource rules; the chart derives the per-role verbs, so do
  # NOT set verbs here (they are ignored). Mutually exclusive with
  # existingClusterRoleName.
  # Example:
  # managedResources:
  # - apiGroups: [""]
  #   resources: ["persistentvolumeclaims", "persistentvolumes", "configmaps", "secrets", "services", "serviceaccounts"]
  # - apiGroups: ["apps"]
  #   resources: ["deployments", "statefulsets", "daemonsets", "replicasets"]
  managedResources: []
  # Name of an existing ClusterRole to bind to the resourcebackup,
  # resourcerestore, and resourcedelete ServiceAccounts. Must include every
  # verb each Job needs (see the "external" mode description above for the
  # required verb union). Mutually exclusive with managedResources.
  existingClusterRoleName: ""
  # Note: under custom or external mode the controller-manager auto-
  # impersonates the <namePrefix>-resourcebackup ServiceAccount on live
  # resource-collection paths (e.g. ResourceViewer with no snapshot/backup
  # ref), so those results honor the scoped Job ClusterRole instead of the
  # controller's wildcard role. Audit logs will show
  # impersonatedUser=system:serviceaccount:<ns>:<sa>. On clusters that deny
  # the impersonate verb to the controller (e.g. hardened admission policy
  # stripping the chart-rendered impersonate rules), live collection will
  # fail with 403 — there is no automatic fallback to the controller's own
  # credentials. Either leave the chart-rendered impersonate rules in place
  # or use the default (full) mode on such clusters.
